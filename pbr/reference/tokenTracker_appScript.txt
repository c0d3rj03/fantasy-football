function grabDailyMFLTransactions() {

  // Refresh Waiver Claims
  let pbrSpreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  var pbrSheet = pbrSpreadsheet.getSheetByName("Waivers");

  var cell = pbrSheet.getRange("E1");
 
  var refresh = parseInt(cell.getValue().toString());
  var increment = refresh + 1;
 
  cell.setValue(increment);
  
  // getTrades()
  // Read transactions page (trades only) from league site and convert to array

  var rawFetch = UrlFetchApp.fetch("https://www.myfantasyleague.com/2026/export?TYPE=transactions&L=63213&APIKEY=PBRreader&W=&TRANS_TYPE=TRADE&JSON=1");
  let tradesJsonString = rawFetch.getContentText();
  let tradesJson = JSON.parse(tradesJsonString);
  var tradesList = tradesJson["transactions"]["transaction"];
  // Now have an object array of trades


  // If only one transaction, it is not an array of objects, just one object
  // Will need to put that object into array by itself
  // Also confirm there hasn't been any trades
  if (tradesList !== undefined && !Array.isArray(tradesList))
  {
    let oneTrade = tradesList;
    tradesList = [];
    tradesList.push(oneTrade);
  }
  // Read league summary from league site and convert to array
  rawFetch = UrlFetchApp.fetch("https://www.myfantasyleague.com/2026/export?TYPE=league&L=63213&JSON=1");
  let leagueInfoJsonString = rawFetch.getContentText();
  let leagueInfoJson = JSON.parse(leagueInfoJsonString);
  var leagueInfo = leagueInfoJson.league;
  var franchiseInfo = leagueInfo["franchises"]["franchise"];
  // Now have an object array of franchise information

  // Extract trade details:
  // - Date of Trade
  // - Team #1 Name
  // - Team #1 Assets Sent
  // - Team #2 Name
  // - Team #2 Assets Sent
  // Will have format of:
  // franchise2_gave_up: "14847,FP_0004_2022_2,FP_0003_2023_2,FP_0003_2023_3,"
  // franchise2:"0003",
  // timestamp:"1607277246"
  // franchise1_gave_up:"14136,FP_0006_2021_5,FP_0004_2021_6,"
  // franchise:"0004"
  // type:"TRADE"
  // comments:""
  // expires:"1607878800"

  // Establish trade asset values
  let tradeValues = new Object();

  // Create object with 2 arrays with trade values
  // Read values for player-traded assets
  pbrSheet = pbrSpreadsheet.getSheetByName("PlayerValues");
  tradeValues.players = pbrSheet.getRange("A2:Q").getValues();

  // Read values for draft pick-traded assets
  pbrSheet = pbrSpreadsheet.getSheetByName("DraftPickValues");
  tradeValues.picks = pbrSheet.getRange("A9:N").getValues();

  // Navigate to Trades sheet.
  pbrSheet = pbrSpreadsheet.getSheetByName("Trades");
  var totalRange = pbrSheet.getRange("A:J")
  totalRange.clear().clearFormat().setFontFamily("Arial Narrow").setFontWeight(10);

  // Create trade sheet header row
  // var tradesValues = pbrSheet.getRange("A1:J1").getValues();
  var tradeSheetDetails = [["Date","Team #1","Assets Sent","Value","","Team #2","Assets Sent","Value","Trade Total Value","Token Each Team"]];

  if (tradesList !== undefined) {

    tradesList.forEach(trade => {

      // Create trade header row
      // Column A - date, Column B - team 1 name, Column F - team 2 name
      tradeRow = ["","","","","","","","","",""];
      var tradeDate = new Date(trade.timestamp*1000);
      var tradeMonth = tradeDate.getMonth()+1;
      tradeRow[0]=tradeDate.toLocaleDateString();
      tradeRow[1]=franchiseInfo[Number(trade.franchise)-1].name;
      tradeRow[5]=franchiseInfo[Number(trade.franchise2)-1].name;
      tradeSheetDetails.push(tradeRow);

      // Format the trade header row
      var topLeft = pbrSheet.getRange("A1");
      var headerRange = topLeft.offset(tradeSheetDetails.length-1,0,1,8);
      headerRange.setBackground("burlywood").setFontWeight("bold");
      headerRange = topLeft.offset(tradeSheetDetails.length-1,8,1,2);
      headerRange.setBackground("lightgreen").setFontWeight("bold");

      // Create assets rows
      // Break up assets from both teams into array 
      var trades1 = trade.franchise1_gave_up.split(",");
      var trades2 = trade.franchise2_gave_up.split(",");

      // length of arrays are one higher than the number of assets
      // starting with asset #1, create trade row and continue until max assets reached
      let assetNo = 0;

      // Accumulate total value from both teams
      totalValue1 = 0;
      totalValue2 = 0;

      while (assetNo < Math.max(trades1.length-1,trades2.length-1)) {
      // Column C - team 1 asset name
      // Column D - team 1 asset value
      // Column G - team 2 asset name
      // Column H - team 2 asset name
        tradeRow = ["","","","","","","","","",""];
        if (assetNo < trades1.length-1) {
          assetInfo = fetchAsset(tradeValues,tradeMonth,trades1[assetNo]);
          tradeRow[2] = assetInfo.name;
          totalValue1 += assetInfo.value;
          tradeRow[3] = assetInfo.value.toFixed(2);
        }
        if (assetNo < trades2.length-1){
          assetInfo = fetchAsset(tradeValues,tradeMonth,trades2[assetNo]);
          tradeRow[6]= assetInfo.name;
          totalValue2 += assetInfo.value;
          tradeRow[7]= assetInfo.value.toFixed(2);
        }
        tradeSheetDetails.push(tradeRow);
        assetNo++;
      }
      // put total value for both teams back on header row for trade
      let tradeHeaderRow = tradeSheetDetails.length-Math.max(trades1.length-1,trades2.length-1)-1;
      tradeSheetDetails[tradeHeaderRow][3]=totalValue1;
      tradeSheetDetails[tradeHeaderRow][7]=totalValue2;
      // and total trade value for both teams
      tradeSheetDetails[tradeHeaderRow][8]=totalValue1+totalValue2;
      // and token count
      tradeSheetDetails[tradeHeaderRow][9]=(Math.floor((totalValue1+totalValue2)/7500)+1)*2;

    });

  };

  // Navigate to Trades sheet. Dump values
  pbrSheet = pbrSpreadsheet.getSheetByName("Trades");
  let sheetTradeRange = pbrSheet.getRange(1, 1, tradeSheetDetails.length, 10);
  sheetTradeRange.setValues(tradeSheetDetails);

  // Format cells
  // Header row
  var range = pbrSheet.getRange("A1:H1");
  range.setBackground("lightsteelblue").setFontWeight("bold");
  var range = pbrSheet.getRange("I1:J1");
  range.setBackground("plum").setFontWeight("bold");

}

function fetchAsset(values,month,asset) {
  // function will return an object that contains a string and value of the asset
  // a. "14847" - Player. 
  //   - From player-value table, look up player number, Return "pos, name, team" and value 
  // b. DP_4_9 - (current) draft pick (5.09) ** zero-based for draft round 
  //   - From DPV table, look up draft pick asset, Return name (Year-current pick) and value  
  // c. FP_0001_2021_2 - FP-future (draft) pick _ team # _ year _ round
  //   - From draftpick-value table, look up draft pick asset, Return name (Year, Round) and value  
  // d. BB_350 - (blind bid) FAAB $350
  //   - From FAAB table, Return "(dollar amount) FAAB" and value
  //   - Asset detail cell: "Blind Bid" and amount
  //   - Asset value cell: Divide amount by 50. Integer.
  //   - Remember amount (and location) in case other side has traded FAAB. Higher and difference is listed

  // Check for type of asset. Return info in object
  let assetDetail = new Object();
  let assetType=asset.substring(0,2);

  if (assetType=="DP") { // DP_2_12
    var pickInfo = values.picks.find(pick => pick[0]==asset);
    assetDetail.name = "Year " + (new Date().getFullYear()) + " Draft Pick " + pickInfo[1];
    assetDetail.value = pickInfo[month+1]; 
  }
  else if (assetType=="FP") { //FP_0007_2023_3
    pickInfo = values.picks.find(pick => pick[0]==(asset.substring(8,12)+" FP_"+asset.slice(-1)));
    assetDetail.name = "Year " + asset.substring(8,12) + " " + pickInfo[1] + " Draft Pick";
    assetDetail.value = pickInfo[month+1]; 
  } 
  else if (assetType=="BB") {
    assetDetail.name = "FAAB $"+asset.substring(3);
    // Make $100 FAAB be 1000 value - like a late 3rd
    assetDetail.value = Math.floor(asset.substring(3)*10);
  }
  else {
    var pickInfo = values.players.find(pick => pick[0]==asset);
    assetDetail.name = pickInfo[2]+" "+pickInfo[4]+", "+pickInfo[3];
    assetDetail.value = pickInfo[month+4]; 
  }
  return assetDetail;
}
